cmake_minimum_required(VERSION 3.24)
project(PerAspera VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASPERA_DESKTOP "Build editor and graphical player" ON)
option(ASPERA_CORE_ONLY "Build dependency-free voxel tools/tests only" OFF)
include(CTest)
add_library(aspera_voxel src/voxel.cpp src/vox_import.cpp src/landscape.cpp)
target_include_directories(aspera_voxel PUBLIC include)
add_executable(aspera_cooker apps/cooker.cpp)
target_link_libraries(aspera_cooker PRIVATE aspera_voxel)
if(BUILD_TESTING)
  add_executable(voxel_tests tests/voxel_tests.cpp)
  target_link_libraries(voxel_tests PRIVATE aspera_voxel)
  add_test(NAME voxel COMMAND voxel_tests)
  add_executable(landscape_tests tests/landscape_tests.cpp)
  target_link_libraries(landscape_tests PRIVATE aspera_voxel)
  add_test(NAME landscape COMMAND landscape_tests)
endif()
if(NOT ASPERA_CORE_ONLY)
  include(cmake/Dependencies.cmake)
  find_package(Threads REQUIRED)
  add_library(aspera_engine src/engine.cpp src/structure.cpp src/physics.cpp src/persistence.cpp src/script.cpp src/network.cpp)
  target_include_directories(aspera_engine PUBLIC include PRIVATE src)
  target_link_libraries(aspera_engine PUBLIC aspera_voxel PRIVATE Jolt enet aspera_lua Threads::Threads)
  add_executable(aspera_server apps/server.cpp)
  target_link_libraries(aspera_server PRIVATE aspera_engine)
  if(BUILD_TESTING)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE aspera_engine)
    add_test(NAME engine COMMAND engine_tests)
    set_tests_properties(engine PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    add_executable(structure_performance_tests tests/structure_performance_tests.cpp
      tests/structure_reference.cpp)
    target_include_directories(structure_performance_tests PRIVATE src tests)
    target_link_libraries(structure_performance_tests PRIVATE aspera_engine)
    add_test(NAME structure_performance COMMAND structure_performance_tests)
    add_executable(landscape_engine_tests tests/landscape_engine_tests.cpp)
    target_link_libraries(landscape_engine_tests PRIVATE aspera_engine)
    target_include_directories(landscape_engine_tests PRIVATE src)
    add_test(NAME landscape_engine COMMAND landscape_engine_tests)
    set_tests_properties(landscape_engine PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 90)
    add_executable(network_tests tests/network_tests.cpp)
    target_link_libraries(network_tests PRIVATE aspera_engine)
    add_test(NAME network COMMAND network_tests)
    set_tests_properties(network PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 30 RUN_SERIAL TRUE)
  endif()
  if(ASPERA_DESKTOP)
    if(BUILD_TESTING)
      add_executable(render_benchmark_tests tests/render_benchmark_tests.cpp)
      target_include_directories(render_benchmark_tests PRIVATE include src)
      add_test(NAME render_benchmark COMMAND render_benchmark_tests)
    endif()
    add_library(aspera_app src/app.cpp src/render.cpp src/render_gi.cpp)
    target_link_libraries(aspera_app PUBLIC aspera_engine PRIVATE SDL3::SDL3 bgfx bx aspera_imgui)
    target_compile_definitions(aspera_app PRIVATE ASPERA_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders")
    if(APPLE)
      set(SHADER_PLATFORM osx)
      set(SHADER_PROFILE metal)
      set(SHADER_BACKEND metal)
    elseif(WIN32)
      set(SHADER_PLATFORM windows)
      set(SHADER_PROFILE s_5_0)
      set(SHADER_BACKEND dx11)
    else()
      set(SHADER_PLATFORM linux)
      set(SHADER_PROFILE spirv)
      set(SHADER_BACKEND spirv)
    endif()
    set(SHADER_OUTPUTS)
    file(GLOB SHADER_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/assets/shaders/[vfc]s_*.sc")
    file(GLOB SHADER_INCLUDES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/assets/shaders/*.sh")
    foreach(SHADER_SOURCE IN LISTS SHADER_SOURCES)
      get_filename_component(STEM "${SHADER_SOURCE}" NAME_WE)
      if(STEM MATCHES "^vs_")
        set(SHADER_TYPE vertex)
      elseif(STEM MATCHES "^cs_")
        set(SHADER_TYPE compute)
      else()
        set(SHADER_TYPE fragment)
      endif()
      set(SHADER_OUT "${CMAKE_BINARY_DIR}/shaders/${SHADER_BACKEND}/${STEM}.bin")
      add_custom_command(OUTPUT "${SHADER_OUT}"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/shaders/${SHADER_BACKEND}"
        COMMAND $<TARGET_FILE:shaderc> -f "${CMAKE_SOURCE_DIR}/assets/shaders/${STEM}.sc" -o "${SHADER_OUT}" --type ${SHADER_TYPE} --platform ${SHADER_PLATFORM} --profile ${SHADER_PROFILE} --varyingdef "${CMAKE_SOURCE_DIR}/assets/shaders/varying.def.sc" -i "${bgfx_cmake_SOURCE_DIR}/bgfx/src"
        DEPENDS shaderc "assets/shaders/${STEM}.sc" assets/shaders/varying.def.sc ${SHADER_INCLUDES} VERBATIM)
      list(APPEND SHADER_OUTPUTS "${SHADER_OUT}")
    endforeach()
    add_custom_target(aspera_shaders DEPENDS ${SHADER_OUTPUTS})
    add_dependencies(aspera_app aspera_shaders)
    install(DIRECTORY "${CMAKE_BINARY_DIR}/shaders" DESTINATION share/aspera)
    add_executable(aspera_editor apps/editor.cpp)
    add_executable(aspera_player apps/player.cpp)
    target_link_libraries(aspera_editor PRIVATE aspera_app)
    target_link_libraries(aspera_player PRIVATE aspera_app)
    install(TARGETS aspera_editor aspera_player RUNTIME DESTINATION bin)
  endif()
  install(TARGETS aspera_server RUNTIME DESTINATION bin)
  install(DIRECTORY "${CMAKE_BINARY_DIR}/licenses" DESTINATION share/aspera)
endif()
install(TARGETS aspera_cooker RUNTIME DESTINATION bin)
install(DIRECTORY games DESTINATION share/aspera)
